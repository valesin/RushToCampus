extends SummerProbeBase
func _ready() -> void:
	await super._ready()
	var game = load("res://scenes/cycling/CyclingGame.tscn").instantiate()
	game.score_path = "user://cycling_probe.json"
	get_tree().root.add_child.call_deferred(game)
	await get_tree().process_frame
	await settle()
	var checks: Dictionary = load("res://tests/cycling/runner_checks.gd").run(game)
	for label in checks:
		report(label, checks[label])
	game.start_run()
	game.traffic.enabled = false
	for spec in [["bus",0,35.0],["car",1,45.0],["car",2,55.0],["cyclist",3,25.0],["pedestrian",4,30.0],["barrier",2,30.0]]:
		game.traffic.actors.append(game.traffic.make_actor(spec[0],spec[1],spec[2]))
	for i in 30:
		await get_tree().physics_frame
	save_frame("art_frame_1")
	for i in 16:
		await get_tree().physics_frame
	save_frame("art_frame_2")
	report("art_loaded",game.presentation.atlas_frames.size() == 2 and game.presentation.city_texture != null)
	await press("cycling_pause",50)
	var d: float = game.distance
	for i in 20:
		await get_tree().physics_frame
	report("art_pause_freezes",game.distance == d)
	await press("cycling_pause",50)
	game.traffic.actors.append(game.traffic.make_actor("car",3,game.distance))
	for i in 50:
		await get_tree().physics_frame
	report("results_after_delay",game.state == game.RunState.RESULTS)
	save_frame("results")
	await press("cycling_confirm",50)
	report("retry_via_button_A",game.state == game.RunState.RUNNING and game.distance < 2.0)
	finish()
